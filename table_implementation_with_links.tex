% 1. Kaggle CT Kidney Dataset
\textbf{\href{https://www.kaggle.com/datasets/nazmul0087/ct-kidney-dataset-normal-cyst-tumor-and-stone}{Kaggle CT Kidney Dataset}} \newline (Dhaka Hospital) & 
CT (Contrast \& Non-contrast) &

Total: 12,446 images \newline Normal: 5,077 \newline Cyst: 3,709 \newline Tumor: 2,283 \newline Stone: 1,377 &

\cite{dataset_kaggle}, \cite{hossain2025detection}, \cite{macin2025kidneynext}, \cite{main23}, \cite{tawfik2026pcsa}, \cite{islam2022vision}, \cite{rathva2025ensemble}, \cite{tarul2025empirical}, \cite{pimpalkar2025fine}, \cite{khan2025multi}, \cite{hossain2024adaptive}, \cite{bhandari2023exploring}, \cite{sharma2025hybrid}, \cite{sharma22}, \cite{martin2025evaluation}, \cite{rajendran2025kidney}, \cite{verma2025automated}, \cite{verma23} \\

% 2. Roboflow Tumor & Cyst Datasets
\textbf{\href{https://universe.roboflow.com/tezskb/kidney-tumor-uqpis}{Roboflow Tumor}} \& \textbf{\href{https://universe.roboflow.com/tezskb/kidney-cyst-6ytvf}{Cyst Datasets}} & 
CT (Axial Slices) & 
Renal Tumor, Renal Cyst (Bounding Boxes \& Polygons) & 
Specialized annotated cohorts for lesion detection and instance segmentation & 
Powers YOLOv8/v11 segmentation modules; achieves instance confidence bounds up to 0.95 (tumors) and 0.89 (cysts) & 
\cite{dataset_roboflow_tumor}, \cite{dataset_roboflow_cyst} \\
\hline

% 3. KiTS19 Challenge
\textbf{\href{https://github.com/neheller/kits19/tree/master/data}{KiTS19 Challenge}} \newline (Tumor Seg. 2019) & 
Corticomedullary CT & 
Kidney, Tumor (2 Seg. Classes) & 
300 CT cases total \newline 210 training / 90 testing & 
Kidney Dice: up to 0.974 \newline Tumor Dice: 0.615 to 0.846 \newline Composite Dice: 0.907 across standard U-Net/PDN baselines & 
\cite{ji2024kits}, \cite{hatsutani2023segmentation}, \cite{abdelrahman2023efficientnet} \\
\hline

% 4. KiTS21 Challenge

Kidney, Tumor, Cyst (3 Seg. Classes) & 
300 CT cases (Official training cohort with multi-annotator consensus) & 
Kidney Dice: up to 0.968 \newline Tumor Dice: up to 0.829 \newline Cyst Dice: up to 0.701; establishes 3D boundary segmentation baselines & 
\cite{heller2023kits21}, \cite{zhao2023automatic}, \cite{chen20215d}, \cite{gohil2024kits} \\

KAUH: 8,400 images \newline \href{https://www.kaggle.com/datasets/orvile/axial-ct-imaging-dataset-kidney-stone-detection}{Sulaimani}: 3,364 slices \newline \href{https://www.cancerimagingarchive.net/collection/ct-colonography/}{TCIA}: 551 CT scans & 
Classification accuracy reaching 98.29\%; volumetric stone detection sensitivity reaching 88.1\% under non-contrast scans & 
\cite{aljohani2025clinical}, \cite{alzubi2022kidney}, \cite{abdalla2025kidney}, \cite{elton2022deep} \\
