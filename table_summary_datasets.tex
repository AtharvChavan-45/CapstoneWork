% 2. Roboflow Universe
\textbf{Roboflow Universe}\newline
\textit{CT (Axial Slices)} & 
Renal Tumor, Renal Cyst (Bounding Boxes \& Polygon Masks) & 
Specialized annotated cohorts for lesion detection and instance segmentation & 
Powers YOLOv8/v11 segmentation modules; achieves instance confidence bounds of 0.95 (tumors) and 0.89 (cysts). & 
\cite{dataset_roboflow_tumor, dataset_roboflow_cyst} \\
\midrule

% 3. KiTS19 Challenge
\textbf{KiTS19 Challenge}\newline
\textit{Corticomedullary CT} & 
Kidney, Tumor (2 Seg. Classes) & 
300 CT cases total (210 train / 90 test) & 
Kidney Dice up to 0.974; Tumor Dice 0.615--0.846; Composite Dice 0.907 across standard U-Net/PDN baselines. & 
\cite{ji2024kits, hatsutani2023segmentation, abdelrahman2023efficientnet} \\
\midrule

% 4. KiTS21 Challenge
\textbf{KiTS21 Challenge}\newline
\textit{Corticomedullary CT} & 
Kidney, Tumor, Cyst (3 Seg. Classes) & 
300 CT cases (consensus ground truth) & 
Kidney Dice up to 0.968; Tumor Dice up to 0.829; Cyst Dice up to 0.701; establishes 3D boundary segmentation baselines. & 
\cite{heller2023kits21, zhao2023automatic, chen20215d, gohil2024kits} \\
\midrule

% 5. Clinical Hospital CT Cohorts
\textbf{KAUH \& Clinical CT}\newline
\textit{Abdominal CT} & 
Kidney Stones, Masses, Tumors, Healthy Tissue & 
KAUH: 8,400 images; Sulaimani: 3,364 slices; TCIA: 551 scans & 
Classification accuracy reaching 98.29\%; volumetric stone detection sensitivity up to 88.1\% under low contrast. & 
\cite{aljohani2025clinical, alzubi2022kidney, abdalla2025kidney, elton2022deep} \\
\midrule

% 6. FLARE Organ Benchmark
\textbf{FLARE Challenge}\newline
\textit{Abdominal CT} & 
Multi-organ segmentation (Kidneys, Liver, Spleen) & 
Low-GPU memory multi-center abdominal CT cases & 
Kidney Dice: 77.8\%--85.2\%; establishes memory-efficient fast abdominal organ localization standards. & 
\cite{ma2022fast} \\
\midrule

% 7. Multi-Center US Repository
\textbf{Multi-Center US}\newline
\textit{Ultrasound (B-mode)} & 
Cysts, Stones, Tumors (3 Classes) & 
$>$10,000 multi-center ultrasound frames & 
Peak F1-score of 0.95; provides cross-modality comparative baseline (US vs. CT) and Grad-CAM interpretability. & 
\cite{obaid2025noisy} \\
\midrule

% 8. Epidemiological Registries
\textbf{NHANES \& SEER}\newline
\textit{Surveys \& Records} & 
Calculi Prevalence (NHANES); RCC Survival (SEER) & 
NHANES: 12,110 participants; SEER: $>$22,000 RCC records & 
Establishes epidemiological baseline for stone prevalence (8.8\%) and survival prognostication AUC of 0.746. & 
\cite{scales2012prevalence, wang2022web} \\
\midrule

\cite{gu2021domain, lee_biobert, fang2023bioformer, luo2022biogpt, liu2024bootstrapping} \\
